Component Purpose File

Registration Processor Handle user signup & account creation signup_process.php

1️⃣ Database - Users Table
SQL Schema - Create Users Table

Table Structure:

CREATE TABLE IF NOT EXISTS users ( user_id INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Unique user ID', email VARCHAR(255) NOT NULL UNIQUE COMMENT 'Email (unique)', password_hash VARCHAR(255) NOT NULL COMMENT 'Bcrypt hashed password', first_name VARCHAR(100) NOT NULL COMMENT 'First name', last_name VARCHAR(100) COMMENT 'Last name', phone_number VARCHAR(20) COMMENT 'Phone number', user_role ENUM('resident','bhw','admin') DEFAULT 'resident', is_active TINYINT(1) DEFAULT 1 COMMENT '1=active, 0=inactive', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT 'Account created', updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, last_login TIMESTAMP NULL COMMENT 'Last login time', INDEX idx_email (email), INDEX idx_role (user_role), INDEX idx_created_at (created_at) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
