```php
<?php
session_start();


if (!isset($_SESSION['user_role']) || $_SESSION['user_role'] !== 'admin') {
    header('Location: login.php');
    exit;
}

require_once 'db_config.php';
?>
<!DOCTYPE html>
<html>
<head>
    <title>Admin Dashboard</title>
    <style>
        body { font-family: Arial; margin: 20px; background: #f5f5f5; }
        .container { max-width: 1200px; margin: 0 auto; background: white; padding: 20px; border-radius: 8px; }
        h1 { color: #454B1B; }
        table { width: 100%; border-collapse: collapse; margin-top: 20px; }
        th, td { padding: 12px; text-align: left; border-bottom: 1px solid #ddd; }
        th { background: #454B1B; color: white; }
        tr:hover { background: #f9f9f9; }
        .admin-badge { background: #dc3545; color: white; padding: 4px 8px; border-radius: 3px; font-size: 12px; }
        .resident-badge { background: #17a2b8; color: white; padding: 4px 8px; border-radius: 3px; font-size: 12px; }
        .bhw-badge { background: #28a745; color: white; padding: 4px 8px; border-radius: 3px; font-size: 12px; }
    </style>
</head>
<body>
    <div class="container">
        <h1>Admin Dashboard - Activity Logs</h1>
        
        <table>
            <thead>
                <tr>
                    <th>Log ID</th>
                    <th>User Name</th>
                    <th>Email</th>
                    <th>Role</th>
                    <th>Action</th>
                    <th>IP Address</th>
                    <th>Timestamp</th>
                </tr>
            </thead>
            <tbody>
                <?php
                $pdo = getDatabaseConnection();
                
                if ($pdo) {
                    $query = "SELECT 
                        al.log_id,
                        al.user_id,
                        u.email,
                        CONCAT(u.first_name, ' ', u.last_name) as user_name,
                        u.user_role,
                        al.action_type,
                        al.ip_address,
                        al.timestamp
                    FROM activity_logs al
                    INNER JOIN users u ON al.user_id = u.user_id
                    ORDER BY al.timestamp DESC
                    LIMIT 100";
                    
                    $stmt = $pdo->prepare($query);
                    $stmt->execute();
                    
                    while ($log = $stmt->fetch()) {
                        $role_class = 'user-' . $log['user_role'] . '-badge';
                        if ($log['user_role'] === 'admin') {
                            $role_badge = '<span class="admin-badge">ADMIN</span>';
                        } elseif ($log['user_role'] === 'bhw') {
                            $role_badge = '<span class="bhw-badge">BHW</span>';
                        } else {
                            $role_badge = '<span class="resident-badge">RESIDENT</span>';
                        }
                        
                        echo "<tr>
                            <td>{$log['log_id']}</td>
                            <td>{$log['user_name']}</td>
                            <td>{$log['email']}</td>
                            <td>{$role_badge}</td>
                            <td><strong>" . strtoupper(str_replace('_', ' ', $log['action_type'])) . "</strong></td>
                            <td><code>{$log['ip_address']}</code></td>
                            <td>" . date('M d, Y H:i:s', strtotime($log['timestamp'])) . "</td>
                        </tr>";
                    }
                }
                ?>
            </tbody>
        </table>
    </div>
</body>
</html>
```
