Unified Navigation Component - Ready to Deploy

Purpose: This is a fully responsive, self-contained navigation bar component for ScanCare dashboards.

Features:

  • ✓ Responsive design: Horizontal menu on desktop (768px+), hamburger toggle on mobile
  • ✓ 4 main navigation items: System Goals, Service Areas, Surveillance, Compliance
  • ✓ Right-aligned user profile dropdown with logout functionality
  • ✓ Sticky positioning for persistent access
  • ✓ Client-side session name management via sessionStorage
  • ✓ Smooth animations and hover states
  • ✓ Inline CSS for reliability - no external dependencies

Integration Instructions:

  1. Copy the entire <nav> block to the top of your dashboard pages.
  2. Ensure the ScanCare logo image exists at images/SClogo.png.
  3. Call updateNavbarName() after user login to display the user's name.
  4. The navbar automatically reads/writes user names from sessionStorage.

JavaScript Functions:

updateNavbarName(userName) - Updates the navbar display name on desktop and mobile menus.

handleLogout() - Clears session data and redirects to login page.

initializeNavbar() - Initializes mobile menu toggle and loads saved username.

function formatUserName(firstName, lastName, middleInitial = '') { let formatted = `${lastName}, ${firstName}`; if (middleInitial && middleInitial.trim()) { const initial = middleInitial.trim().charAt(0).toUpperCase(); formatted += ` ${initial}.`; } return formatted; } function initializeNavbar() { const toggleBtn = document.getElementById('toggleBtn'); const mobileMenuPanel = document.getElementById('mobileMenuPanel'); if (toggleBtn) { toggleBtn.addEventListener('click', function () { const isVisible = mobileMenuPanel.style.display === 'flex'; mobileMenuPanel.style.display = isVisible ? 'none' : 'flex'; toggleBtn.classList.toggle('active', !isVisible); }); } const savedUserName = sessionStorage.getItem('scancare_user_name'); if (savedUserName) { updateNavbarName(savedUserName); } const mobileLinks = mobileMenuPanel?.querySelectorAll('a'); if (mobileLinks) { mobileLinks.forEach(link => { link.addEventListener('click', () => { mobileMenuPanel.style.display = 'none'; if (toggleBtn) toggleBtn.classList.remove('active'); }); }); } } function updateNavbarName(userNameData) { const desktopName = document.getElementById('desktopUserName'); const mobileName = document.getElementById('mobileUserName'); let displayName = ''; if (typeof userNameData === 'object' && userNameData !== null) { displayName = formatUserName( userNameData.firstName || '', userNameData.lastName || '', userNameData.middleInitial || '' ); } else { displayName = String(userNameData); } if (desktopName) desktopName.textContent = displayName; if (mobileName) mobileName.textContent = displayName; sessionStorage.setItem('scancare_user_name', displayName); } function handleLogout() { sessionStorage.removeItem('scancare_user_name'); sessionStorage.removeItem('scancare_user_email'); window.location.href = 'mock-login.html'; } document.addEventListener('DOMContentLoaded', initializeNavbar);
